{{/* Wraps a pricing line in a focusable tooltip and, when a url is given, a link to the docs. Falls back to the plain text when there is neither. Params: text (string), tip (string, optional), url (string, optional). */}} {{ $label := .text | markdownify }} {{ with .url }} {{ $external := or (hasPrefix . "http") (hasPrefix . "//") }} {{ $href := cond $external . (. | relLangURL) }} {{ $attrs := cond $external (printf ` target="_blank" rel="noopener"`) "" }} {{ $label = printf `%s` $href $attrs $label | safeHTML }} {{ end }} {{ if .tip }} {{ $label }} {{ .tip }} {{ else }} {{ $label }} {{ end }}
